/// **A CONTACT ZONE FIRES BECAUSE SHE WALKED INTO IT.**
///
/// ⚠ **`EdgeExit` IS the contact zone in this game.** A census of every authored
/// `.ldtk` finds 381 `Door` zones, 72 `EdgeExit`s and not one `Walk` — so a test
/// that asked for `Walk` would sit on its own "no such zone" guard forever while
/// contact transitions rotted.
///
/// ⭐ **it needs ONE of the room's contact zones to work, not a chosen one.**
/// See `zones_by_distance`: a single walled-off rectangle is level geometry, and
/// only "none of them" is a statement about the mechanism.
#[test]
fn walking_into_a_contact_zone_changes_the_room() {
    let mut sim = fixed_60hz_sim();
    for _ in 0..10 {
        sim.step(base());
    }
    let before = active_room(&mut sim);
    let zones = zones_by_distance(&mut sim, LoadingZoneActivation::EdgeExit);
    // ⚠ LOUD, never a quiet `return`. A test that skips itself when it cannot
    // find its subject reports green for the one reason it exists.
    assert!(
        !zones.is_empty(),
        "'{before}' authors no `EdgeExit` loading zone, so this test walked at \
         nothing — point it at a room that has one",
    );

    let mut report = Vec::new();
    for zone in &zones {
        // ⛔ **A FRESH SESSION PER ZONE, and this is not tidiness.** Continuing
        // from where the last walk stalled starts the next one at the FAR end of
        // the room, so the budget — not the geometry — decides whether she
        // arrives, and an out-of-budget walk is indistinguishable from a wall.
        let mut sim = fixed_60hz_sim();
        for _ in 0..10 {
            sim.step(base());
        }
        let (arrived, after) = walk_into(&mut sim, zone, base());
        if after.is_some() {
            return;
        }
        report.push(format!(
            "  `{}` {:?}: {}",
            zone.name,
            zone.aabb,
            if arrived {
                "she got inside it and the room did not change"
            } else {
                "she never reached it (blocked or out of reach)"
            }
        ));
    }
    let reached_any = report.iter().any(|line| line.contains("got inside"));
    let report = report.join("\n");
    assert!(
        false,
        "walked at every one of the {} contact zones in '{before}' and never \
         left the room:\n{report}\n\n{}",
        zones.len(),
        if reached_any {
            "⛔ she got INSIDE at least one and it did not fire: that is the \
             TRANSITION half — overlap → transition_for_player → \
             RoomTransitionRequested → the room actually changing."
        } else {
            "⛔ she reached none of them, so this is the MOVEMENT half and not \
             contact transitions at all: something between a held direction and \
             the body's position stopped carrying her."
        },
    );
}
